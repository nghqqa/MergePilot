校验方法

# ── 在线拉取核验（以 registry index digest 为准）──
docker pull ghcr.io/nghqqa/mergepilot-console@sha256:aec584ead1314760ab72d021a820ed5cf3ae4f8973df580dd7df252f8dcc5ae3
docker image inspect ghcr.io/nghqqa/mergepilot-console --format '{{range .RepoDigests}}{{println .}}{{end}}'
# RepoDigests 必须含 @sha256:6d25…（它记录"实际拉到的 digest"，与引擎无关）；
# registry 侧可再核对：docker buildx imagetools inspect <tag> 的 Digest 行 = 同值

# ── 离线 tar 核验（文件级 SHA256 为第一权威）──
sha256sum -c SHA256SUMS        # 期望 offline.tar.gz = f896ce53…e2d1
docker load -i mergepilot-console-v0.2.0-beta.6-rc.35.6-rc4-offline.tar.gz
# 加载后 docker image inspect <tag> 的 .Id 应与【同引擎】在线拉取记录的 .Id 一致
#（.Id 口径随引擎存储模式而异，跨引擎不比较 .Id，以 tar SHA256 + index digest 为权威）

# ── 运行版本核验 ──
curl -s http://127.0.0.1:48500/api/health | grep -E "version|git_commit"
# 期望：v0.2.0-beta.6-rc.35.6-rc1 / git_commit=null（本版构建未注入 GIT_COMMIT，
# 源码归属以上方表格与 Release 记载为准）
