校验方法

# 镜像 ID 校验（加载/拉取后）
docker inspect --format '{{.Id}}' <镜像引用>
# 期望：sha256:e30419990434c209de8d9de7fc11fc267d8c9e6fe0d99ff242eda69b7a61dd3a

# 运行版本校验
curl -s http://127.0.0.1:48500/api/health | grep version
# 期望：v0.2.0-beta.6-rc.35.5

# 附件校验
sha256sum -c SHA256SUMS
