# ── 在线拉取核验（以 registry index digest 为准）──
docker pull ghcr.io/nghqqa/mergepilot-console@sha256:f8bc369f5b046536103d75ba53c4863871abb3ecddbdfe920baf29c6428afe9a
docker buildx imagetools inspect ghcr.io/nghqqa/mergepilot-console:v0.2.0-beta.6-rc.35.6-rc8.5
# Digest 行 = sha256:f8bc369f…fe9a（OCI index，linux/amd64 + attestation）

# ── 离线 tar 核验（文件级 SHA256 为第一权威）──
sha256sum -c SHA256SUMS        # 期望 rc8.5-offline.tar.gz = 35decdcf…cbe5
docker load -i mergepilot-console-v0.2.0-beta.6-rc.35.6-rc8.5-offline.tar.gz
# 加载后 docker image inspect 的 .Id = sha256:f8bc369f…fe9a（containerd 存储=index digest；经典存储=config 2b80128b…ec46）

# ── 运行版本核验 ──
curl -s http://127.0.0.1:48500/api/health | grep -E "version|git_commit"
