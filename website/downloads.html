校验方法

# ── 在线拉取核验（以 registry index digest 为准）──
docker pull ghcr.io/nghqqa/mergepilot-console@sha256:c874b4d88ef67387401caf5b04d94ee9d4aa2fd9c6254301fc167a9d5505cbd3
docker buildx imagetools inspect ghcr.io/nghqqa/mergepilot-console:v0.2.0-beta.6-rc.35.6-rc8.4.1
# Digest 行 = sha256:c874b4d8…5cbd3（OCI index，linux/amd64 + attestation）

# ── 离线 tar 核验（文件级 SHA256 为第一权威）──
sha256sum -c SHA256SUMS        # 期望 rc8.4.1-offline.tar.gz = 4d917f3d…da6f
docker load -i mergepilot-console-v0.2.0-beta.6-rc.35.6-rc8.4.1-offline.tar.gz
# 加载后 docker image inspect 的 .Id = sha256:c874b4d8…5cbd3（containerd 存储=index digest；经典存储=config 44206496…0039）

# ── 运行版本核验 ──
curl -s http://127.0.0.1:48500/api/health | grep -E "version|git_commit"
# 期望：v0.2.0-beta.6-rc.35.6-rc8.4.1 / git_commit=ba089624e266f7aa14dda3f45285a3232b7f5938
