# ── 在线拉取核验（以 registry index digest 为准）──
docker pull ghcr.io/nghqqa/mergepilot-console@sha256:003d963347d400222c3b503cef0fe840452d20f1429db78f495e5585f9be3eea

# ── 离线 tar 核验（文件级 SHA256 为第一权威）──

docker load -i mergepilot-console-v0.2.0-beta.6-rc.35.6-rc8.tar.gz
# 加载后 docker image inspect 的 .Id 应 = sha256:c5081829…23de（本版 config digest，经典存储）

# ── 运行版本核验 ──
curl -s http://127.0.0.1:48500/api/health | grep -E "version|git_commit"
# 期望：v0.2.0-beta.6-rc.35.6-rc8 / git_commit=a50f034be27f6fa89ebd8553afee42c674e34ade
